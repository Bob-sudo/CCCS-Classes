\nextq
Complete the code fragment below so as to produce the following output:
\begin{console}[fontsize=\small]
4
 3 2 1 0
3
 2 1 0
2
 1 0
1
 0
\end{console}
\ANSWER
\begin{answercode}
for (int i = 4; i > 0; i--)
{
    std::cout << i << '\n';
    for (int j = i - 1; j >= 0; j--)
    {
        std::cout << ' ' << j;
    }
    std::cout << '\n';
}
\end{answercode}

\nextq
Complete the code fragment below so as to produce the following output:
\begin{console}[fontsize=\small]
4
 1 2 3 4
3
 3 2 1
2
 1 2
1
 1
\end{console}
\ANSWER
\begin{answercode}
for (int i = 4; i > 0; i--)
{
    std::cout << i << '\n';
    if (i % 2 == 1) // CASE: i is odd. Print j running from i to 1
    {
        for (int j = i; j > 0; j--)
        {
            std::cout << ' ' << j;
        }
    }
    else // CASE: i is even. Print j running up 1 to i
    {
        for (int j = 1; j <= i; j++)
        {
            std::cout << ' ' << j;
        }
    }
    std::cout << '\n';
}
\end{answercode}
