What is the output of this program:
\begin{console}[fontsize=\small]
int i = 2;
int j = 1;
while (i < 5)
{
    j += i;
    std::cout << i << ' ' << j << std::endl;
    ++i;
}
std::cout << i << ' ' << j << std::endl;
\end{console}

3 6
4 10
5 10

\end{answercode}
