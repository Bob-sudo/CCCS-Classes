\nextq
Write a C++ program that declares an array \verb$a$ of
1000 doubles \textit{initializing} all the values to 0.0,
obtain 1000 doubles from the user to be placed in the array (using a loop),
compute and print the average of the values in the array (using another loop),
and finally compute and print the average of the values in the array which are
strictly positive (i.e., $> 0$) (using yet another loop).
\\
\ANSWER
\begin{answercode}
#include <iostream>

int main()
{
    double a[1000] = {0.0};

    for (int i = 0; i < 1000; ++i)
    {
        std::cin >> a[i];
    }

    double total_sum = 0.0;
    for (int i = 0; i < 1000; ++i)
    {
        total_sum += a[i];
    }
    double overall_average = total_sum / 1000.0;
    std::cout << overall_average << '\n';

    double positive_sum = 0.0;
    int positive_count = 0;
    for (int i = 0; i < 1000; ++i)
    {
        if (a[i] > 0)
        {
            positive_sum += a[i];
            positive_count++;
        }
    }

    if (positive_count > 0)
    {
        double positive_average = positive_sum / positive_count;
        std::cout << positive_average << '\n';
    }
    else
    {
        std::cout << "No positive values in the array.\n";
    }

    return 0;
}
\end{answercode}
