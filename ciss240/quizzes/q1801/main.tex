\verb!target! appears in string \verb!s!.

\begin{console}[fontsize=\footnotesize,commandchars=\\\{\}]
\userinput{apple_pies}

\end{console}
\\
\ANSWER
\begin{answercode}
#include <iostream>

int main()
{
    char s[1024];
    std::cin >> s; 
    char target;
    std::cin >> target; 

    int count = 0;

    // TODO
    for (int i = 0; s[i] != '\0'; ++i)
    {
        if (s[i] == target)
        {
            count++;
        }
    }

    std::cout << count << '\n';
    return 0;
}

\end{answercode}
