\nextq
Complete the following template swap, bubblesort, and println functions.
(Bubblesort should use the swap function.)
\\
\ANSWER
\begin{answercode}
#include <iostream>

void swap(T & x, T & y)
{
  T temp = x;
  x = y;
  y = temp;
}

void bubblesort(T x[], int size)
{
  for (int i = 0; i < size - 1; ++i)
  {
    for (int j = 0; j < size - 1 - i; ++j)
    {
      if (x[j] > x[j + 1])
      {
        swap(x[j], x[j + 1]);
      }
    }
  }
}

void println(T x[], int size)
{
    for (int i = 0; i < size; ++i)
    {
        std::cout << x[i] << ' ';
    }
}

int main()
{
    int x[] = {5, 3, 1, 2};
    bubblesort< int >(x, 4);            // the < int > is not necessary
    println< int >(x, 4);               // the < int > is not necessary

    double y[] = {5.5, 3.3, 1.1, 2.2};
    bubblesort< double >(y, 4);         // the < double > is not necessary
    println< double >(y, 4);            // the < double > is not necessary

    return 0;
}
\end{answercode}
