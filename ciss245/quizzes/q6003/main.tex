\nextq
A hunter lives in the Wumpus world, which is a 4-by-4 grid.
He/she walks in the direction he/she is facing.
Initially he/she is facing N (north).
In each time step in Wumpus world, the hunter
(1) randomly selects a direction,
(2) randomly decides to walk one step (in his/her direction) or not,
and
(3) randomly decides to fire one arrow (in his/her direction) or not.
Here's a test run:
\begin{Verbatim}[frame=single,fontsize=\footnotesize]
+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
|H| | | |
+-+-+-+-+
Hunter direction: N

+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| |H| | |
+-+-+-+-+
Hunter direction: E

+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| |H| | |
+-+-+-+-+
Hunter direction: S

+-+-+-+-+
| | | | |
+-+-+-+-+
| |^| | |
+-+-+-+-+
| |H| | |
+-+-+-+-+
| | | | |
+-+-+-+-+
Hunter direction: N

+-+-+-+-+
| |^| | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| | |H| |
+-+-+-+-+
| | | | |
+-+-+-+-+
Hunter direction: E

+-+-+-+-+
| | | | |
+-+-+-+-+
| | | | |
+-+-+-+-+
| | | |H|
+-+-+-+-+
| | | | |
+-+-+-+-+
Hunter direction: E
\end{Verbatim}
The \verb!^! is the arrow flying in the N direction.
You should draw the arrow using
\verb!^!,
\verb!v!,
\verb!>!,
\verb!<! depending on the direction
N,S,E,W (respectively)
of flight.

Note that when the hunter fires his/her arrow, the arrow will fly in the
direction of the hunter (at that point in time).
The hunter can only fire one arrow (too bad).
The hunter will also not walk into his arrow.

You are given this:
\begin{Verbatim}[frame=single,fontsize=\footnotesize]
// file: WumpusWorld.h
#ifndef WUMPUSWORLD_H
#define WUMPUSWORLD_H

#include <iostream>

class WumpusWorld
{
public:
    void init();         // initialize so that hunter is at (3,0).
    void println();      // print according to the above format
    void move_hunter();  // 1. randomize a direction rand() % 4 where 0,1,2,3
                         //    are N,S,E,W for hunter.
                         // 2. randomize whether to walk or not
                         // 3. randomize whether to fire arrow or not
                         // Of course hunter must stay in the world.
                         // If hunter tries to move N but its row is 0,
                         // hunter stays put.
    void move_arrow();   // move arrow (if necessary)
private:
    char world_[4][4];
    int hunter_direction_;
    int arrow_state_;    // 0-N, 1-S, 2-E, 3-W, 4-not fired, 5-hit something
};
#endif
\end{Verbatim}

You want to make the hunter wonder around like this:
\begin{Verbatim}[frame=single,fontsize=\footnotesize]
#include <iostream>
#include <ctime>
#include <cstdlib>
#include "WumpusWorld.h"

int main()
{
    srand((unsigned int) time(NULL));

    WumpusWorld ww;
    ww.init();
    ww.println();
    for (int i = 0; i < 50; ++i)
    {
        ww.move_hunter();
        ww.move_arrow();
        ww.println();
    }
    return 0;
}
\end{Verbatim}
Complete the following cpp file that contains the implementation
of the methods declared in the WumpusWorld class.
\\
\ANSWER
\begin{answercode}
// file: WumpusWorld.cpp
int hunter_row, hunter_col;
int arrow_row, arrow_col;
int arrow_dir;

void WumpusWorld::init()
{
  for (int i = 0; i < 4; ++i)
  {
    for (int j = 0; j < 4; ++j)
    {
      world_[i][j] = ' ';
    }
  }

  hunter_row = 3;
  hunter_col = 0;
  hunter_direction_ = 0;

  world_[hunter_row][hunter_col] = 'H';

  arrow_state_ = 4;
}

void WumpusWorld::println()
{
  for (int i = 0; i < 4; ++i)
  {
    std::cout << "+-+-+-+-+" << std::endl;
    for (int j = 0; j < 4; ++j)
    {
      std::cout << "|" << world_[i][j];
    }
    std::cout << "|" << std::endl;
  }
  std::cout << "+-+-+-+-+" << std::endl;

  std::string dir;
  switch (hunter_direction_)
  {
    case 0: dir = "N"; break;
    case 1: dir = "S"; break;
    case 2: dir = "E"; break;
    case 3: dir = "W"; break;
  }
  std::cout << "Hunter direction: " << dir << std::endl;
}

void WumpusWorld::move_hunter()
{
  hunter_direction_ = rand() % 4;

  if (rand() % 2 == 1)
  {
    int new_row = hunter_row;
    int new_col = hunter_col;

    if (hunter_direction_ == 0 && hunter_row > 0) new_row--;
    else if (hunter_direction_ == 1 && hunter_row < 3) new_row++;
    else if (hunter_direction_ == 2 && hunter_col < 3) new_col++;
    else if (hunter_direction_ == 3 && hunter_col > 0) new_col--;

    if (world_[new_row][new_col] != '^' &&
    world_[new_row][new_col] != 'v' &&
    world_[new_row][new_col] != '<' &&
    world_[new_row][new_col] != '>')
    {
      world_[hunter_row][hunter_col] = ' ';
      hunter_row = new_row;
      hunter_col = new_col;
      world_[hunter_row][hunter_col] = 'H';
    }
  }

  if (arrow_state_ == 4 && rand() % 2 == 1)
  {
    arrow_state_ = 0; 
    arrow_row = hunter_row;
    arrow_col = hunter_col;
    arrow_dir = hunter_direction_;

    switch (arrow_dir)
    {
      case 0: world_[arrow_row][arrow_col] = '^'; break;
      case 1: world_[arrow_row][arrow_col] = 'v'; break;
      case 2: world_[arrow_row][arrow_col] = '>'; break;
      case 3: world_[arrow_row][arrow_col] = '<'; break;
    }
  }
}

void WumpusWorld::move_arrow()
{
  if (arrow_state_ != 0) return;

  if (!(arrow_row == hunter_row && arrow_col == hunter_col))
  world_[arrow_row][arrow_col] = ' ';

  if (arrow_dir == 0) arrow_row--;
  else if (arrow_dir == 1) arrow_row++;
  else if (arrow_dir == 2) arrow_col++;
  else if (arrow_dir == 3) arrow_col--;

  if (arrow_row < 0 || arrow_row > 3 || arrow_col < 0 || arrow_col > 3)
  {
    arrow_state_ = 5;
    return;
  }

  switch (arrow_dir)
  {
    case 0: world_[arrow_row][arrow_col] = '^'; break;
    case 1: world_[arrow_row][arrow_col] = 'v'; break;
    case 2: world_[arrow_row][arrow_col] = '>'; break;
    case 3: world_[arrow_row][arrow_col] = '<'; break;
  }
}

\end{answercode}
