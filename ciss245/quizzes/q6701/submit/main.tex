\nextq
Correct the code below so that it produces the following output:
\begin{Verbatim}[frame=single,fontsize=\footnotesize]
<Werewolf john>
<Lucan john>
\end{Verbatim}
\begin{answercode}
#include <iostream>
#include <string>

class Lucan
{
public:
    Lucan(const std::string name)
        : name_(name)
    {}

    std::string name_;
};

std::ostream & operator<<(std::ostream & cout, const Lucan & l)
{
    cout << "<Lucan " << l.name_ << '>';
    return cout;
}

class Werewolf
{
public:
    Werewolf(const std::string name)
        : name_(name)
    {}

    operator Lucan() const
    {
        return Lucan(name_);
    }

    std::string name_;
};

std::ostream & operator<<(std::ostream & cout, const Werewolf & w)
{
    cout << "<Werewolf " << w.name_ << '>';
    return cout;
}

int main()
{
    Werewolf w("john");
    std::cout << w << '\n';
    std::cout << Lucan(w) << '\n';
    return 0;
}
\end{answercode}
