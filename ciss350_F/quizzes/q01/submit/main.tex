\nextq
Write a function \verb!replace()! that replaces a character in a C-string
with another.
For instance suppose you have
\vspace{-2mm}
\begin{Verbatim}[frame=single,fontsize=\small]
char x[1024] = "helloworld";
\end{Verbatim}
\vspace{-5mm}
If you call
\vspace{-3mm}
\begin{Verbatim}[frame=single,fontsize=\small]
replace(x, 'o', 'a');
\end{Verbatim}
\vspace{-4mm}
and then you print \verb!x!, the output is
\vspace{-3mm}
\begin{Verbatim}[frame=single,fontsize=\small]
hellawarld
\end{Verbatim}
\ANSWER
\begin{answercode}
#include <iostream>

void replace(char * x, char src, char dest)
{
  for(int i = 0; x[i] != '\0'; i++)
  {
    if (x[i] == src)
        x[i] = dest;
  }
}

int main()
{
    char x[1024];
    char src, dest;
    std::cin >> x >> src >> dest;
    replace(x, src, dest);
    std::cout << '[' << x << "]\n";
    return 0;
}
\end{answercode}

\nextq
Write a function \verb!remove()! that removes all occurrences of a
character from a C-string.
For instance suppose you have
\vspace{-2mm}
\begin{Verbatim}[frame=single,fontsize=\small]
char x[1024] = "helloworld";
\end{Verbatim}
\vspace{-5mm}
If you call
\vspace{-3mm}
\begin{Verbatim}[frame=single,fontsize=\small]
remove(x, 'o');
\end{Verbatim}
\vspace{-4mm}
and then you print \verb!x!, the output is
\vspace{-3mm}
\begin{Verbatim}[frame=single,fontsize=\small]
hellwrld
\end{Verbatim}
\ANSWER
\begin{answercode}
#include <iostream>

void remove(char * x, char c)
{
  int j = 0;
  for(int i = 0; x[i] != '\0'; i++)
  {
    if (x[i] != c)
    {
      x[j++] = x[i];
    }
  }
  x[j] = '\0';
}

int main()
{
    char x[1024];
    char c;
    std::cin >> x >> c;
    remove(x, c);
    std::cout << '[' << x << "]\n";
    return 0;
}
\end{answercode}
